```sql
-- Tabela para códigos de vinculação
CREATE TABLE IF NOT EXISTS whatsapp_link_codes (
  id INT PRIMARY KEY AUTO_INCREMENT,
  user_id INT NOT NULL,
  code VARCHAR(20) NOT NULL UNIQUE,
  whatsapp_number VARCHAR(20) NOT NULL,
  used BOOLEAN DEFAULT false,
  expires_at TIMESTAMP NOT NULL,
  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  INDEX idx_code (code),
  INDEX idx_whatsapp (whatsapp_number),
  INDEX idx_expires (expires_at)
);

-- Tabela para vinculações ativas
CREATE TABLE IF NOT EXISTS whatsapp_links (
  id INT PRIMARY KEY AUTO_INCREMENT,
  user_id INT NOT NULL UNIQUE,
  whatsapp_number VARCHAR(20) NOT NULL UNIQUE,
  linked_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  last_interaction TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  is_active BOOLEAN DEFAULT true,
  FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
  INDEX idx_user (user_id),
  INDEX idx_whatsapp (whatsapp_number)
);
```

```javascript
      // ============================================
      // 🤖 SISTEMA DE COMANDOS BOT
      // ============================================
      sock.ev.on('messages.upsert', async ({ messages, type }) => {
        for (const msg of messages) {
          // Ignorar mensagens enviadas por você
          if (msg.key.fromMe) continue;
          
          const from = msg.key.remoteJid.replace('@s.whatsapp.net', '');
          const text = msg.message?.conversation || 
                       msg.message?.extendedTextMessage?.text;
          
          if (!text) continue;
          
          const command = text.trim().toLowerCase();
          
          console.log(`📩 Mensagem recebida de ${from}: ${command}`);
          
          try {
            // ========== COMANDO: !vincular ==========
            if (command === '!vincular') {
              // Verificar se já está vinculado
              const existingLink = await database.query(
                'SELECT user_id FROM whatsapp_links WHERE whatsapp_number = ? AND is_active = true',
                [from]
              );
              
              if (existingLink.length > 0) {
                const user = await database.query(
                  'SELECT username, email FROM users WHERE id = ?',
                  [existingLink[0].user_id]
                );
                
                await sock.sendMessage(msg.key.remoteJid, {
                  text: `✅ *WhatsApp já vinculado!*\n\n` +
                        `👤 Usuário: ${user[0].username}\n` +
                        `📧 Email: ${user[0].email}\n\n` +
                        `Digite *!menu* para ver os comandos disponíveis.`
                });
                continue;
              }
              
              // Gerar código único
              const code = `MOZH-${Math.random().toString(36).substr(2, 6).toUpperCase()}`;
              const expiresAt = new Date(Date.now() + 15 * 60 * 1000); // 15 minutos
              
              // Salvar código temporário (sem user_id ainda)
              await database.query(
                'INSERT INTO whatsapp_link_codes (user_id, code, whatsapp_number, expires_at, used) VALUES (0, ?, ?, ?, false)',
                [code, from, expiresAt]
              );
              
              await sock.sendMessage(msg.key.remoteJid, {
                text: `🔐 *Código de Vinculação MozHost*\n\n` +
                      `Seu código: *${code}*\n\n` +
                      `⏰ Válido por 15 minutos\n\n` +
                      `📱 Acesse o painel MozHost e digite este código na seção "Vincular WhatsApp"\n\n` +
                      `🔗 https://mozhost.topaziocoin.online/whatsapp-link`
              });
            }
            
            // ========== COMANDO: !menu ==========
            else if (command === '!menu' || command === '!ajuda' || command === '!help') {
              await sock.sendMessage(msg.key.remoteJid, {
                text: `🤖 *MozHost Bot - Comandos*\n\n` +
                      `*Vinculação:*\n` +
                      `!vincular - Gerar código de vinculação\n\n` +
                      `*Conta:*\n` +
                      `!saldo - Ver seus coins\n` +
                      `!plano - Ver detalhes do plano\n\n` +
                      `*Containers:*\n` +
                      `!containers - Listar containers\n` +
                      `!status - Status dos containers\n\n` +
                      `*Pagamento:*\n` +
                      `!pagamento - Comprar coins\n` +
                      `!faturas - Ver faturas\n\n` +
                      `*Suporte:*\n` +
                      `!suporte - Falar com suporte\n` +
                      `!ajuda - Ver este menu`
              });
            }
            
            // ========== COMANDO: !saldo ==========
            else if (command === '!saldo') {
              const link = await database.query(
                'SELECT user_id FROM whatsapp_links WHERE whatsapp_number = ? AND is_active = true',
                [from]
              );
              
              if (link.length === 0) {
                await sock.sendMessage(msg.key.remoteJid, {
                  text: `❌ WhatsApp não vinculado!\n\nEnvie *!vincular* para vincular sua conta.`
                });
                continue;
              }
              
              const user = await database.query(
                'SELECT username, coins FROM users WHERE id = ?',
                [link[0].user_id]
              );
              
              await sock.sendMessage(msg.key.remoteJid, {
                text: `💰 *Saldo de Coins*\n\n` +
                      `👤 ${user[0].username}\n` +
                      `🪙 Coins: *${user[0].coins}*\n\n` +
                      `Digite *!pagamento* para comprar mais coins.`
              });
            }
            
            // ========== COMANDO: !containers ==========
            else if (command === '!containers' || command === '!status') {
              const link = await database.query(
                'SELECT user_id FROM whatsapp_links WHERE whatsapp_number = ? AND is_active = true',
                [from]
              );
              
              if (link.length === 0) {
                await sock.sendMessage(msg.key.remoteJid, {
                  text: `❌ WhatsApp não vinculado!\n\nEnvie *!vincular* para vincular sua conta.`
                });
                continue;
              }
              
              const containers = await database.query(
                'SELECT name, status, domain FROM containers WHERE user_id = ? ORDER BY created_at DESC LIMIT 10',
                [link[0].user_id]
              );
              
              if (containers.length === 0) {
                await sock.sendMessage(msg.key.remoteJid, {
                  text: `📦 *Seus Containers*\n\nVocê ainda não tem containers.\n\nAcesse o painel para criar: https://mozhost.topaziocoin.online`
                });
              } else {
                const statusEmoji = { running: '🟢', stopped: '🔴', error: '⚠️', starting: '🟡' };
                const containerList = containers.map(c => 
                  `${statusEmoji[c.status] || '⚪'} *${c.name}*\n   Status: ${c.status}\n   ${c.domain ? `🔗 ${c.domain}` : ''}`
                ).join('\n\n');
                
                await sock.sendMessage(msg.key.remoteJid, {
                  text: `📦 *Seus Containers*\n\n${containerList}`
                });
              }
            }
            
            // ========== COMANDO: !pagamento ==========
            else if (command === '!pagamento' || command === '!comprar') {
              await sock.sendMessage(msg.key.remoteJid, {
                text: `💳 *Comprar Coins*\n\n` +
                      `*Pacotes disponíveis:*\n\n` +
                      `🪙 100 Coins - 100 MT\n` +
                      `🪙 500 Coins - 450 MT (10% desconto)\n` +
                      `🪙 1000 Coins - 800 MT (20% desconto)\n\n` +
                      `📱 Acesse o painel para comprar:\n` +
                      `https://mozhost.topaziocoin.online/billing`
              });
            }
            
            // ========== COMANDO DESCONHECIDO ==========
            else if (text.startsWith('!')) {
              await sock.sendMessage(msg.key.remoteJid, {
                text: `❌ Comando não reconhecido.\n\nDigite *!menu* para ver os comandos disponíveis.`
              });
            }
            
          } catch (error) {
            console.error('❌ Erro ao processar comando:', error);
            await sock.sendMessage(msg.key.remoteJid, {
              text: `❌ Erro ao processar comando. Tente novamente.`
            });
          }
        }
      });
```
